    ```dart
    testWidgets('starts ticker when engine is idle', (tester) async {

    });
    ```

    **测试 3: engineState=playing → ticker stops**
    ```dart
    testWidgets('stops ticker when engine is playing', (tester) async {
      final engineState = ValueNotifier(MediaState.idle);
      await tester.pumpWidget(MaterialApp(
        home: Scaffold(body: AuroraBackground(engineState: engineState)),
      ));
      engineState.value = MediaState.playing;
      await tester.pump();
      expect(find.byType(AuroraBackground), findsOneWidget);
      engineState.dispose();
    });
    ```

    **测试 4: engineState transition idle→playing→idle**
    ```dart
    testWidgets('restarts ticker after idle→playing→idle', (tester) async {

      await tester.pumpWidget(MaterialApp(
        home: Scaffold(body: AuroraBackground(engineState: engineState)),
      ));
      engineState.value = MediaState.playing;
      await tester.pump();
      engineState.value = MediaState.idle;

    });
    ```

    **测试 5: dispose cleanup**
    ```dart
    testWidgets('cleans up resources on dispose', (tester) async {
    ```
