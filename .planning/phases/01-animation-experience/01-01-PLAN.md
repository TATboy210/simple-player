    onHover: (event) {
      // Bottom trigger zone: only show when mouse is within 150px of screen bottom
      final bottomTriggerZone = Tokens.controlBarHeight +
          Tokens.controlBarMarginBottom + 24;
      // event.position.dy is relative to the MouseRegion which starts at
      // (controlBarMarginBottom + controlBarHeight) from bottom.
      // We need to check if mouse is in the bottom portion of this region.
      final regionHeight = MediaQuery.of(context).size.height -
          (Tokens.controlBarMarginBottom + Tokens.controlBarHeight);

    },
    ```

    void _handleTap() {
      if (_clickTimer?.isActive ?? false) {
        // Second tap within 250ms → double-click, toggle fullscreen
        _clickTimer?.cancel();
        widget.actions.onToggleFullscreen?.call();
      } else {
        // First tap → immediately hide (no delay)
        _clickTimer = Timer(
          const Duration(milliseconds: ControlsOverlay._clickDelayMs),
          () {
            // Timer fires after 250ms if no second tap — timer is now just
            // a cooldown to prevent the next single-tap from hiding during
            // double-tap detection window. Hide happens NOW.
          },
        );
        // Immediate hide on first tap
        if (widget.engine.state.value == MediaState.idle) return;
        _autoHide.hide();
      }
    }
    ```
