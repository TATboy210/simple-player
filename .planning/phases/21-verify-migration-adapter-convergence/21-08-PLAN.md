```
group('KernelAdapter routing', () {
  // Setup: 两个独立 FakeEngine 实例（legacy 和 migrated）
  late FakeEngine legacyEngine;
  late FakeEngine migratedEngine;

  group('per-method routing via migratedMethods', () {
    // 验证 _targetFor(method) 正确路由
    test('method in migratedMethods routes to migrated engine')
    test('method NOT in migratedMethods routes to legacy engine')
    test('empty migratedMethods routes all methods to legacy')
    test('all27 methods in migratedMethods routes all to migrated')
  })

  group('capability field routing', () {
    // 验证7个 per-capability 字段路由
    test('stateView=legacy returns legacy.state')
    test('stateView=migrated returns migrated.state')
    test('volume=legacy returns legacy.volume')
    test('volume=migrated returns migrated.volume')
    // ... 类似测试其他5个字段: playback, track, subtitle, videoEffect, renderer
  })

  group('DelegationPolicy.all() constructor', () {
    test('all(legacy) sets all7 fields to legacy')
    test('all(migrated) sets all7 fields to migrated')
    test('all() clears migratedMethods to empty set')
  })

  group('dispose behavior', () {
    test('dispose calls stateView-target engine dispose')
    test('dispose calls bundle.dispose()')
  })

  group('DiagnosticsBundle forwarding', () {
    test('bundle is accessible via adapter')
  })
})
```

- 验证 ValueNotifier identity forwarding（ADAPT-03）：adapter.state 返回的 notifier 与目标引擎的 notifier 是同一实例
