**HIGH-2: Fix _clampToVisibleBounds to check all displays**

Replace `_clampToVisibleBounds` (lines 496-515) with a version that checks all connected displays:

```dart
Future<void> _clampToVisibleBounds(Size savedSize, Offset savedPosition) async {
  try {
    const double minVisible = 100;
    final views = ui.PlatformDispatcher.instance.views;

    // Check if window is visible on ANY display (per HIGH-2)
    bool isOnAnyDisplay = false;
    for (final view in views) {
      final screenW = view.physicalSize.width / view.devicePixelRatio;
      final screenH = view.physicalSize.height / view.devicePixelRatio;

      final isOnThisDisplay = savedPosition.dx + savedSize.width >= minVisible &&
          savedPosition.dy + savedSize.height >= minVisible &&
          savedPosition.dx <= screenW - minVisible &&
          savedPosition.dy <= screenH - minVisible;

      if (isOnThisDisplay) {
        isOnAnyDisplay = true;
        break;
      }
    }

    if (!isOnAnyDisplay) {
      await windowManager.center();
    }
  } on Exception catch (e) {
    debugPrint('[WindowManager] bounds check failed: $e');
  }
}
```

Note: This still has the limitation that `FlutterView` doesn't expose per-display position. The logical bounds of each view start at (0,0). For true multi-monitor support, we'd need `window_manager`'s `getMonitorInfo()`. But the loop structure is correct — when that API becomes available, only the inner check needs updating.

```dart
class _TitleBarButtonState extends State<_TitleBarButton> {
  bool _hovered = false;

  @override
  void initState() {
    super.initState();
    PlatformService.I.isResizing.addListener(_onResizingChanged);
  }

  @override
  void dispose() {
    PlatformService.I.isResizing.removeListener(_onResizingChanged);
    super.dispose();
  }

  void _onResizingChanged() {
    // MEDIUM-1: When resize ends, re-evaluate hover state
    if (!PlatformService.I.isResizing.value && mounted) {
      // Re-check if cursor is over this button by forcing a rebuild
      // The MouseRegion onEnter/onExit will fire on next mouse move
      // But we need to clear stale _hovered state from before resize
      setState(() => _hovered = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    final bgColor = _hovered
        ? (widget.isClose ? Tokens.danger : Tokens.bgHover)
        : Colors.transparent;
    final iconColor = _hovered
        ? Tokens.textPrimary
        : (widget.isActive ? Tokens.accent : Tokens.textSecondary);

    return Tooltip(
      message: widget.tooltip,
      waitDuration: const Duration(milliseconds: 400),
      child: MouseRegion(
        cursor: SystemMouseCursors.click,
        onEnter: (_) {
          if (!PlatformService.I.isResizing.value) {
            setState(() => _hovered = true);
          }
        },
        onExit: (_) {
          if (!PlatformService.I.isResizing.value) {
            setState(() => _hovered = false);
          }
        },
        child: GestureDetector(
          behavior: HitTestBehavior.opaque,
          onTap: widget.onPressed,
          child: Container(
            width: Tokens.titleBarButtonWidth,
            height: Tokens.titleBarHeight,
            color: bgColor,
            child: Icon(widget.icon, size: Tokens.iconSm, color: iconColor),
          ),
        ),
      ),
    );
  }
}
```
