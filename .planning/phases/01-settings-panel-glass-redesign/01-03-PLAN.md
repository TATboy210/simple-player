```dart
  child: Column(
    crossAxisAlignment: CrossAxisAlignment.start,
    children: [
      SectionHeader(title: l10n.shortcutsTab, icon: Icons.keyboard),
      for (final def in defs)
        SettingActionRow(
          label: def.label,
          valueText: friendlyKeyName(_currentKeyFor(def.action, def.defaultKey)),
          isActive: _recordingAction == def.action,
          activeText: l10n.pressKeyToBind,
          onAction: () => _startRecording(def.action),
          onDeactivate: _recordingAction == def.action ? _cancelRecording : null,
        ),
    ],
  ),
)
```
