cmake_minimum_required(VERSION 3.14)
project(fullscreen_resize_guard_test LANGUAGES CXX)
enable_testing()
add_executable(fullscreen_resize_guard_test fullscreen_resize_guard_test.cpp)
target_compile_features(fullscreen_resize_guard_test PRIVATE cxx_std_17)
target_compile_definitions(fullscreen_resize_guard_test PRIVATE UNICODE _UNICODE)
target_compile_options(fullscreen_resize_guard_test PRIVATE /utf-8 /W4 /WX)
target_link_libraries(fullscreen_resize_guard_test PRIVATE comctl32)
add_test(NAME fullscreen_resize_guard COMMAND fullscreen_resize_guard_test)
